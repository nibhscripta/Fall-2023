    \begin{align*}
        \text{function} &= e^{-x/2} \\
        \text{operator} &= \frac{d^2}{dx^2} - x^2 \\

    \end{align*}

    This function is an eigenfunction.

    $\boxed{\text{eigenvalue} = -16}$

        \frac{1}{\sin\theta} \frac{d}{d\theta}\left(\sin\theta 
        \frac{d}{d\theta}\right) \left(\sin\theta \cos\theta\right) &= \frac{1}{\sin\theta} \frac{d}{d\theta} \left(\sin\theta\cos^2\theta - \sin^2\theta\right) \\

    \end{align*}

    $\boxed{\text{Not an eigenfunction.}}$

    \item Problem 2 (3.1)

        \int_{-\infty}^{\infty} \psi^*\psi dx &= \int_{-\infty}^{\infty} e^{-2bx^2} dx \\
        &= 2 \cdot \int_{0}^{\infty} e^{-2bx^2} dx \\

        \Aboxed{\psi_{\text{norm}} &= \left(\frac{2b}{\pi}\right)^{1/4} e^{-bx^2}}
