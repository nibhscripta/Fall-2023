    \item \textbf{Problem 1 (7-1)}
    
    1$^{\text{st}}$ condition

    \begin{align*}
        P_L &= 20 \text{ mm Hg} = 2666 \text{ Pa} \\
        P_H &= 1 \text{ atm} = 101325 \text{ Pa} \\
        y_0 &= 0.209 \\
        y_j &= 1 \text{ ppm} = 1\cdot10^{-6} \\
        y_j &= y_0 \left(\frac{P_L}{P_H}\right)^j \\
        \ln\left(\frac{y_j}{y_0}\right) &= \ln\left(\left(\frac{P_L}{P_H}\right)^j\right) \\
        j &= \frac{\ln\left(\frac{y_j}{y_0}\right)}{\ln\left(\frac{P_L}{P_H}\right)} \\
        j &= \frac{\ln\left(\frac{1\cdot10^{-6}}{0.209}\right)}{\ln\left(\frac{2666}{101325}\right)} \\
        \Aboxed{j &= 3.37} \\
        \Delta n &= j \left(P_H - P_L\right) \frac{V}{RT} \\
        \Delta n &= 4 \cdot \left(101325 - 2666\right)  \cdot \frac{3.78}{8.314 \cdot 297} \\
        \Delta n &= 603.8 \text{ mol} \\
        m &= \frac{603.8 \cdot 28}{1000} \\
        \Aboxed{m &= 16.9 \text{ kg}}
    \end{align*}

    2$^{\text{nd}}$ condition

    \begin{align*}
        P_L &= 30 \text{ mm Hg} = 4000 \text{ Pa} \\
        P_H &= 1 \text{ atm} = 101325 \text{ Pa} \\
        y_0 &= 0.209 \\
        j &= \frac{\ln\left(\frac{1\cdot10^{-6}}{0.209}\right)}{\ln\left(\frac{4000}{101325}\right)} \\
        \Aboxed{j &= 3.79} \\
        \Delta n &= j \left(P_H - P_L\right) \frac{V}{RT} \\
        \Delta n &= 4 \cdot \left(101325 - 4000\right)  \cdot \frac{37}{8.314 \cdot 297} \\
        \Delta n &= 5830.4 \text{ mol} \\
        m &= \frac{5830.4 \cdot 28}{1000} \\
        \Aboxed{m &= 163.3 \text{ kg}}
    \end{align*}
